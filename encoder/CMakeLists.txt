cmake_minimum_required(VERSION 3.16)
project(blp_encoder VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "编码组件仅提供 Windows x64 构建。")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BLP_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(TURBOJPEG_DIR "${BLP_ROOT}/third_party/turbojpeg")
add_library(blp_encoder SHARED blp_encoder.cpp "${BLP_ROOT}/src/blp/blp_codec.cpp")
target_compile_definitions(blp_encoder PRIVATE BLP_ENCODER_BUILD NOMINMAX)
target_include_directories(blp_encoder PRIVATE "${BLP_ROOT}/src" "${BLP_ROOT}/third_party/stb" "${TURBOJPEG_DIR}/include")
target_link_libraries(blp_encoder PRIVATE "${TURBOJPEG_DIR}/lib/turbojpeg-static.lib")
if(MSVC)
    target_compile_options(blp_encoder PRIVATE /utf-8 /W4)
endif()
option(BLP_ENCODER_BUILD_TESTS "构建 C ABI 验证程序" ON)
if(BLP_ENCODER_BUILD_TESTS)
    enable_testing()
    add_executable(blp_encoder_api_test "${BLP_ROOT}/tests/blp_encoder_api_test.c")
    target_include_directories(blp_encoder_api_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(blp_encoder_api_test PRIVATE blp_encoder)
    if(MSVC)
        target_compile_options(blp_encoder_api_test PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME blp_encoder_api COMMAND blp_encoder_api_test)
endif()
install(TARGETS blp_encoder RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES blp_encoder.h DESTINATION include)
install(FILES "${BLP_ROOT}/LICENSE" DESTINATION licenses RENAME BLP-Viewer-LICENSE)
install(FILES "${BLP_ROOT}/third_party/stb/stb_image_resize2-LICENSE.txt" DESTINATION licenses RENAME stb-LICENSE.txt)
install(FILES "${TURBOJPEG_DIR}/LICENSE.md" DESTINATION licenses RENAME libjpeg-turbo-LICENSE.md)
install(FILES "${TURBOJPEG_DIR}/README.ijg" DESTINATION licenses)
install(FILES README.md DESTINATION .)
